# gguf publishers worth watching. the first three are the ones MODELS.md
# already rates for quality; the rest are tracked because the trending list
# keeps surfacing them, not because they are endorsed.
#
# the `[role]` markers are read by ./propose, which is why they are here rather
# than in a comment: whether a repo is somebody's re-quant of a model that
# already exists is decided by WHO OWNS IT, and that is a fact this file holds.
# asking a model to infer it from the repo name got `bartowski/Qwen2.5-Coder-
# 32B-Instruct-GGUF` classified as an original release by the lab that trained
# it. a list beats a guess, and the guess is not even cheaper.

[quantizer]
# these publish other people's models. a repo here is a quant decision, never a
# roster one, whatever the name looks like
unsloth             # UD-* dynamic quants, measurably ahead at equal file size
bartowski           # consistent imatrix calibration, the reliable fallback
mradermacher        # i1-* covers the long tail, least benchmarked of the three
ggml-org            # upstream reference quants, first to land on new archs
ISTA-DASLab         # ist austria's quantization lab. publishes the measurement
                    # WITH the weights -- a per-tensor allocation dump, the
                    # imatrix, and a benchmark table that runs its own rungs
                    # against unsloth's at matched file size, which is the only
                    # published comparison of two quant families on one sweep
cstr                # the speech half's quantizer: crispasr's own README names a
                    # cstr repo for 95 of its backends, and until this line the
                    # catalog swept none of them -- so the conversions were
                    # invisible to discovery AND their `base_model` was
                    # unavailable, which is the join that says which weights a
                    # conversion converts

[vendor]
# model authors who publish their own gguf. a new repo here can be a release
AngelSlim           # tencent's quant team (hy3)
AtomicChat          # ling 3.0 flash, where no unsloth repo exists
LiquidAI
meta-models
ibm-granite         # publishes gguf for every granite size it releases
microsoft           # fara, and the phi line before it

[community]
# community fine-tunes and requantizations. high download counts, no
# independent evaluation -- tracked for discovery, not recommended
agentionai          # AP-* ladders of qwen3.8, plus its own Signal fine-tunes.
                    # the two roles sit in one account, so the per-repo
                    # `base_model:` tag decides which a repo is and this marker
                    # only says where to look
DavidAU             # multi-stage merges, NEO/MAX imatrix variants
huihui-ai           # abliterated variants
prism-ml
ornith-ai
antirez             # hand-tuned deepseek v4 quants
